function scoreAverage(a){

return a.reduce((s,x)=>s+x.score,0)/a.length

}

function classification(boat,a){

if(!x)return{type:"normal",diff:0,average:0};

let avg=scoreAverage(a),

low[0].boat===boat

return{type:"normal",diff,average:avg}

}
