function ticketRoleMultiplier(x){
if(!x||!x.third||x.box44||x.box75)return 1;
const aiRank=b=>Math.max(1,aiPos(b)||6);
let mult=1;
/* AI期待値：1着を最重視、2着、3着の順に役割別で差をつける */
mult*=1+(7-aiRank(x.first))*.075;
mult*=1+(7-aiRank(x.second))*.045;
mult*=1+(7-aiRank(x.third))*.025;
/* AI評価1位＋今レース攻め手を積極的に着へ絡める */
let evalOrder=[1,2,3,4,5,6].sort((a,b)=>score(b)-score(a)),eval1=evalOrder[0];
if(x.first===eval1)mult*=1.18;if(x.second===eval1)mult*=1.10;if(x.third===eval1)mult*=1.06;
if(x.first===attacker)mult*=1.20;if(x.second===attacker)mult*=1.12;if(x.third===attacker)mult*=1.08;
if(eval1===attacker){if(x.first===attacker)mult*=1.16;else if(x.second===attacker)mult*=1.10;else if(x.third===attacker)mult*=1.06}
/* 1着のAI決まり手ランキング適合を強く評価 */
if(x.type==="escape"){
if(course(x.first)===1){let er=outlook?.escape||0;mult*=er>=65?1.36:er>=55?1.24:er>=45?1.12:1.22}
}else{
let same=nonEscapeTop10.find(z=>z.boat===x.first&&z.type===x.type);
if(same){let r=nonEscapeTop10.indexOf(same)+1;mult*=1+Math.max(0,11-r)*.032}
}
/* 2着が決まり手定石にどれだけ合うかを明確に反映 */
if(x.type&&x.type!=="escape"&&x.type!=="box"){
let pool=[1,2,3,4,5,6].filter(b=>b!==x.first)
.map(b=>({b,v:secondConventionScore(x.first,b,x.type)})).sort((a,b)=>b.v-a.v);
let r=pool.findIndex(z=>z.b===x.second);
if(r===0)mult*=1.32;else if(r===1)mult*=1.18;else if(r===2)mult*=1.07;else mult*=.91
}
/* 3着はAI期待値＋3連対適性。裏返しただけの組み合わせにも差が出る */
let thirdRank=aiRank(x.third);
if(thirdRank===1)mult*=1.20;else if(thirdRank===2)mult*=1.13;else if(thirdRank===3)mult*=1.07;else if(thirdRank>=5)mult*=.94;

mult*=Math.max(.88,Math.min(1.18,.94+t3/500));
return mult
}

function dedupePatternGroups(groups,sections){
const uniq=a=>[...new Set((a||[]).filter(Boolean))];
const patternRows=p=>groups.filter(x=>x.pattern===p);
const axisSig=rows=>{
if(!rows.length)return"";
let first=rows[0].first,kind=rows[0].kind||"";
let seconds=uniq(rows.map(x=>x.second)).sort((a,b)=>a-b);
let thirds=uniq(rows.map(x=>x.third)).sort((a,b)=>a-b);
return `${kind}|${first}|${seconds.join(".")}|${thirds.join(".")}`
};
const priorityFor=sample=>{
let base=[];
if(sample?.type==="escape")base=escapeDecisionRanking(rankings).map(x=>x.boat);
else base=nonEscapeDecisionRanking().map(x=>x.boat);
return uniq([...base,...aiAll.map(x=>x.boat),1,2,3,4,5,6])
};
const baseRaw=x=>{
if(x.kind==="2連単")return Math.max(.1,aiScore(x.first)*.70+aiScore(x.second)*.30+motorRate(x.first)*.18+motorRate(x.second)*.10);
return Math.max(.1,aiScore(x.first)*.58+aiScore(x.second)*.31+aiScore(x.third)*.18+motorRate(x.first)*.10)
};
const rebuild=(p,newSeconds,newThirds,oldRows)=>{
if(!oldRows.length)return false;
let sample=oldRows[0],first=sample.first,isExacta=sample.kind==="2連単";
let ratios=oldRows.map(x=>x.raw/Math.max(.1,baseRaw(x))).filter(Number.isFinite);
let ratio=ratios.length?ratios.reduce((a,b)=>a+b,0)/ratios.length:1;
let made=[];
for(const second of newSeconds){
if(!second||second===first)continue;
if(isExacta){
let x={...sample,second,third:null};x.raw=baseRaw(x)*ratio;made.push(x);continue
}
for(const third of newThirds){
if(!third||third===first||third===second)continue;
let x={...sample,second,third};x.raw=baseRaw(x)*ratio;made.push(x)
}
}
if(!made.length)return false;
for(let i=groups.length-1;i>=0;i--)if(groups[i].pattern===p)groups.splice(i,1);
groups.push(...made);
return true
};
const nextSet=(original,count,priority,excluded)=>{
let out=[];
for(const b of priority){
if(excluded.includes(b)||original.includes(b)||out.includes(b))continue;
out.push(b);if(out.length>=count)break
}
return out
};
let seen=new Map();
for(const sec of [...sections]){
let rows=patternRows(sec.pattern);if(!rows.length)continue;
let sig=axisSig(rows);
if(!seen.has(sig)){seen.set(sig,sec);continue}
let duplicateOf=seen.get(sig),sample=rows[0],priority=priorityFor(sample);
let first=sample.first,oldSeconds=uniq(rows.map(x=>x.second)),oldThirds=uniq(rows.map(x=>x.third));
let corrected=false,stage="";
let nextSeconds=nextSet(oldSeconds,oldSeconds.length,priority,[first]);
if(nextSeconds.length===oldSeconds.length){
let thirds=oldThirds.filter(b=>!nextSeconds.includes(b)&&b!==first);
while(thirds.length<oldThirds.length){
let b=priority.find(v=>v!==first&&!nextSeconds.includes(v)&&!thirds.includes(v));if(!b)break;thirds.push(b)
}
if(rebuild(sec.pattern,nextSeconds,thirds,rows)&&axisSig(patternRows(sec.pattern))!==sig){corrected=true;stage="2着候補を次順位へ補正"}
}
if(!corrected){
rows=patternRows(sec.pattern);
let seconds=uniq(rows.map(x=>x.second)),thirds=uniq(rows.map(x=>x.third));
let nextThirds=nextSet(thirds,thirds.length,priority,[first,...seconds]);
if(nextThirds.length===thirds.length&&rebuild(sec.pattern,seconds,nextThirds,rows)&&axisSig(patternRows(sec.pattern))!==sig){corrected=true;stage="3着候補を次順位へ補正"}
}
if(corrected){
sec.duplicateAdjusted=true;
sec.note=(sec.note?sec.note+" / ":"")+`🔄 重複補正：${duplicateOf.title}と完全一致したため${stage}`;
let ns=axisSig(patternRows(sec.pattern));
if(!seen.has(ns))seen.set(ns,sec);
continue
}
for(let i=groups.length-1;i>=0;i--)if(groups[i].pattern===sec.pattern)groups.splice(i,1);
let baseRows=patternRows(duplicateOf.pattern);baseRows.forEach(x=>x.logicMatchCount=(x.logicMatchCount||1)+1);
duplicateOf.logicMatch=true;
duplicateOf.note=(duplicateOf.note?duplicateOf.note+" / ":"")+`🔥 複数ロジック一致：${sec.title}と同一買い目のため統合`;
let si=sections.indexOf(sec);if(si>=0)sections.splice(si,1)
}
}

}
