function overallEscapeProbability(){

return{escape:0,nonEscape:100,status:"イン以外",matchAdjustment:0,matchLabel:"1コース不成立"};

p+=(d.escape-50)*.45;

if(d.escape>=83)p+=10;

else if(d.makareRate>=15)p-=8;
else if(d.makareRate>=7)p-=3;

else if(d.sasareRate>=15)p-=7;
else if(d.sasareRate>=7)p-=3;
if(d.makiSashiSareRate<7)p+=3;

}

p+=(o.escape-50)*.22;

let vt=venueTrait();
p+=(vt.escape-1)*35;
let matchup=[];
let maxOutsideAttack=0;
for(let b=2;b<=6;b++){

if(!rb)continue;
let ad=decisions(rb);
let items=[

];
items.forEach(x=>{

}
})
}
matchup.sort((a,b)=>b.penalty-a.penalty);
let matchPenalty=0;
if(matchup[0])matchPenalty+=matchup[0].penalty;
if(matchup[1])matchPenalty+=matchup[1].penalty*.45;
if(matchup[2])matchPenalty+=matchup[2].penalty*.20;
matchPenalty=Math.min(18,matchPenalty);
let noMeaningfulAttack=maxOutsideAttack<15;
let matchBonus=0;
if(noMeaningfulAttack){
matchBonus=p<45?10:6;
p+=matchBonus;
if(d.escape>=40&&p<45)p=45
}
if(matchPenalty>0)p-=matchPenalty;

/* 手動展示評価を全体見通しへ連動。全部「－」なら補正0 */
let exhibitInside=exhibitPoint(1),exhibitOutside=[];
for(let b=2;b<=6;b++)exhibitOutside.push({boat:b,point:exhibitPoint(b)});
let exhibitAdjustment=Math.max(-8,Math.min(8,exhibitInside/5));
let attackExhibit=exhibitOutside.find(x=>x.boat===bestAttackBoat(dominant(overview())))?.point||0;
let strongOutsideExhibit=exhibitOutside.map(x=>x.point).sort((a,b)=>b-a).slice(0,2);
let outsideExhibitPressure=0;
if(attackExhibit>0)outsideExhibitPressure+=Math.min(4,attackExhibit/10);
for(const ep of strongOutsideExhibit)if(ep>0)outsideExhibitPressure+=Math.min(2,ep/20);
outsideExhibitPressure=Math.min(8,outsideExhibitPressure);
exhibitAdjustment-=outsideExhibitPressure;
p+=exhibitAdjustment;

/* SG級など全員の絶対値が高いレースでも、6艇内の相対戦力で逃げ率を補正 */
let insideBase=Math.max(0,score(1)-exhibitPoint(1));
let outsideBase=[2,3,4,5,6].map(b=>({boat:b,value:Math.max(0,score(b)-exhibitPoint(b))}));
let outsideAvg=outsideBase.reduce((s,x)=>s+x.value,0)/(outsideBase.length||1);
let relativeGap=insideBase-outsideAvg,relativeAdjustment=0;

else if(relativeGap>=15)relativeAdjustment+=3;
else if(relativeGap>=5)relativeAdjustment+=0;
else if(relativeGap>=-5)relativeAdjustment-=5;
else if(relativeGap>=-14)relativeAdjustment-=10;
else relativeAdjustment-=18;
let strongerOutside=outsideBase.filter(x=>x.value>=insideBase).length;
if(strongerOutside>=3)relativeAdjustment-=15;
else if(strongerOutside===2)relativeAdjustment-=10;
else if(strongerOutside===1)relativeAdjustment-=5;

if(attackBoat>=2&&attackBoat<=6&&outsideRank>0&&outsideRank<=2)relativeAdjustment-=7;
if(attackBoat>=2&&attackBoat<=6&&exhibitPoint(attackBoat)>=20)relativeAdjustment-=6;
if(insideBase-outsideAvg>=20&&exhibitPoint(1)>=20&&strongerOutside===0)relativeAdjustment+=5;
relativeAdjustment=Math.max(-40,Math.min(11,relativeAdjustment));
p+=relativeAdjustment;

if(matchPenalty>=10&&d.escape<83&&p>=75)p=74.9;
/* 高確率帯は上限で潰さず圧縮して差を残す。75〜85%はそのまま、85%超を段階圧縮 */
let rawEscapeP=p;
if(rawEscapeP>95)p=89.6+(rawEscapeP-95)*.18;
else if(rawEscapeP>85)p=85+(rawEscapeP-85)*.46;
p=Math.max(5,Math.min(93,p));
let matchAdjustment=matchBonus-matchPenalty+exhibitAdjustment+relativeAdjustment;
let baseMatchLabel=noMeaningfulAttack?
`外の攻め手15%以上なし → イン逃げ +${matchBonus.toFixed(1)}pt補正`:
matchup.length?
`${matchup[0].boat}号艇 ${matchup[0].method}${matchup[0].attack.toFixed(1)}% × 1号艇対応弱点${matchup[0].defense.toFixed(1)}% → イン逃げ -${matchPenalty.toFixed(1)}pt補正`:
"決まり手一致の強い攻めなし → 補正なし";
let matchLabel=`${baseMatchLabel} / 展示 ${exhibitAdjustment>=0?"+":""}${exhibitAdjustment.toFixed(1)}pt / 相対戦力 ${relativeAdjustment>=0?"+":""}${relativeAdjustment.toFixed(1)}pt`;
return{
escape:p,
nonEscape:100-p,
matchAdjustment,
matchLabel,
status:
p>=75?
"イン逃げ":
p>=45?
"中間":
"イン以外"
}
}
function dominant(o){
return Object.entries(o)
.sort((a,b)=>b[1]-a[1])[0][0]
}
function typeName(t){
return{
escape:"イン逃げ",
sashi:"差し",
maki:"まくり",
makiSashi:"まくり差し"
}[t]||t
}
function mainTypes(o){
return Object.entries(o)
.sort((a,b)=>b[1]-a[1])
.filter(([t,v])=>t==="escape"?v>=42:v>=20)
}
function score(b){
let r=racer(b),p=preview(b);
if(!r)return 0;
let d=decisions(r);
let s=
num(r.national_win_rate||r.win_rate_national)*7+
num(r.local_win_rate||r.win_rate_local)*4+
num(r.national_top_2_percent||r.national_2ren_rate)*.12;
s+=num(
r.motor_top_2_percent||
r.motor_2ren_rate||
r.motor_2ren
)*.15;
let motor=num(
r.motor_top_2_percent||
r.motor_2ren_rate||
r.motor_2ren
);
s+=
motor>=40?7:
motor>=35?5:
motor>=30?2:
motor<20?-3:0;
let ast=num(r.average_start_timing||r.average_st);
if(ast>0){
s+=Math.max(0,.30-ast)*20;
if(ast<=.15)s+=5;
else if(ast<=.18)s+=3
}
let st=num(p.start_timing||p.st);
if(st>0){
s+=Math.max(0,.25-st)*15;
if(st<=.15)s+=4;
else if(st<=.18)s+=2
}
let ex=num(p.exhibition_time||p.exhibition);
if(ex>0){
s+=Math.max(0,6.8-ex)*5;
if(ex<=6.70)s+=5;
else if(ex<=6.80)s+=3;
else if(ex>=6.95)s-=3
}
s+=({1:10,2:6,3:5,4:3,5:1,6:0}[course(b)]||0);
let hs=hstats(b);
if(hs.count){
s+=hscore(b);
s+=
hs.rate>=60?5:
hs.rate>=40?3:
hs.rate<20?-2:0
}
let form=recentForm(b);
s+=form.score;
let rk=rank(r.rank_number||r.class_number||r.rank||r.class);
s+=
rk==="A1"?5:
rk==="A2"?2:
rk==="B2"?-2:0;
let o=overview(),dom=dominant(o);
if(course(b)!==1)
s+=suitability(b,dom)*.08;
Object.entries(o).forEach(([t,v])=>{
if(t!=="escape"&&v>=20)
s+=suitability(b,t)*.03
});
let vt=venueTrait();
if(course(b)===1){
s+=(vt.escape-1)*20;
s-=d.makare*.20;
s-=d.sasare*.15;
s-=d.makiSashi*.10;
if(ast>0&&ast<=.18)s+=3;
let de=decisionEvaluation(1,"escape");
s+=Math.max(-3,Math.min(6,de.value*.30))
}
if(course(b)===2){
let de=Math.max(
decisionEvaluation(b,"sashi").value,
decisionEvaluation(b,"maki").value
);
s+=Math.max(-2,Math.min(6,de*.22))
}
if(course(b)>=3&&course(b)<=5){
let de=Math.max(
decisionEvaluation(b,"maki").value,
decisionEvaluation(b,"makiSashi").value,
decisionEvaluation(b,"sashi").value
);
s+=Math.max(-2,Math.min(6,de*.22))
}
let wi=windInfo();
if(wi.mode==="head"&&wi.speed>=3){
if(course(b)>=3){
s+=windSuitability(b,"maki")*.20;
s+=windSuitability(b,"makiSashi")*.15
}
if(course(b)===1)s-=2
}
if(wi.mode==="tail"&&wi.speed>=3){
if(course(b)===1)s+=windSuitability(b,"escape")*.25;
if(course(b)===2)s+=windSuitability(b,"sashi")*.20
}
/* 手動展示評価：未入力（－）なら0pt */
s+=exhibitPoint(b);
return Math.max(0,s)
}
function ranking(){
return Array.from({length:6},(_,i)=>({
boat:i+1,
racer:racer(i+1),
score:score(i+1)
})).sort((a,b)=>b.score-a.score)
}
function scoreAverage(a){
return a.reduce((s,x)=>s+x.score,0)/a.length
}
function classification(boat,a){
let sorted=[...a].sort((p,q)=>q.score-p.score),avg=scoreAverage(a),x=sorted.find(z=>z.boat===boat);
if(mandatoryBetProtectedBoats&&mandatoryBetProtectedBoats.has(boat))return{type:"normal",diff:x?x.score-avg:0,average:avg,rankGap:0,inherited:false,baseType:"normal",under100:false,belowAverage:x?x.score<avg:false,mandatoryProtected:true};
if(!x)return{type:"normal",diff:0,average:avg,rankGap:0,inherited:false,baseType:"normal",under100:false,belowAverage:false};
let i=sorted.findIndex(z=>z.boat===boat),diff=x.score-avg,rankGap=i>0?sorted[i-1].score-x.score:0;
/* 絶対条件：AI期待値100点超は3着まで／切り候補にしない */
if(x.score>100)return{type:"normal",diff,average:avg,rankGap,inherited:false,baseType:"normal",under100:false,belowAverage:x.score<avg};
let under100=true,belowAverage=x.score<avg,type="normal";
if(belowAverage&&rankGap>=25)type="cut";
else if(belowAverage&&rankGap>=20)type="third";
return{type,diff,average:avg,rankGap,inherited:false,baseType:type,under100,belowAverage}
}
function strongMotor(b){
let m=num(
racer(b)?.motor_top_2_percent||
racer(b)?.motor_2ren_rate||
racer(b)?.motor_2ren
);
return m>=35
}
function goodHistory(b){
let s=hstats(b);
return s.count>=5&&s.rate>=40
}
function bestAttackBoat(type){
let best=1,val=-Infinity;
for(let b=1;b<=6;b++){
let x=
suitability(b,type)+
recentForm(b).score*.5;
if(x>val){
val=x;
best=b
}
}
return best
}
function developmentBonus(b,rankings,type){
let attacker=bestAttackBoat(type),
ac=course(attacker),
bc=course(b),
bonus=0,
reasons=[];
if(
(type==="maki"||type==="makiSashi")&&
b!==attacker&&
bc>ac
){
if(bc===ac+1){
bonus+=15;
reasons.push("まくり艇直後")
}else if(bc>ac+1){
bonus+=6;
reasons.push("まくり外側展開")
}
}
if(
type==="maki"&&
b!==attacker&&
bc===ac+1
){
bonus+=6;
reasons.push("まくり差し候補位置")
}
if(
type==="makiSashi"&&
b!==attacker&&
bc>ac
){
bonus+=5;
reasons.push("まくり差し展開")
}
if(b===1&&course(1)===1){
let d=decisions(racer(1));
if(d.makare>=15){
bonus+=8;
reasons.push("まくられ残り再評価")
}
if(d.sasare>=15){
bonus+=7;
reasons.push("差され残り再評価")
}
}
if(
windMode()==="head"&&
windInfo().speed>=3&&
(type==="maki"||type==="makiSashi")&&
b!==attacker&&
bc>ac
){
bonus+=6;
reasons.push("向かい風アウト展開")
}
if(
windMode()==="tail"&&
windInfo().speed>=3&&
b===1&&
course(b)===1
){
bonus+=6;
reasons.push("追い風イン有利")
}
if(
windMode()==="tail"&&
windInfo().speed>=3&&
type==="sashi"&&
bc===2
){
bonus+=6;
reasons.push("追い風差し有利")
}
return{bonus,reasons}
}
function promotionByMaki(b,rankings,type){
let attacker=bestAttackBoat(type),
bc=course(b),
ac=course(attacker);
if(attacker===b)return false;
if(
(type==="maki"||type==="makiSashi")&&
bc===ac+1
){
let hs=hstats(b),f=recentForm(b);
return hs.count>=5&&hs.rate>=40||
f.score>=3||
strongMotor(b)||
score(b)>=scoreAverage(rankings)-5
}
return false
}
function attackRevival(b,rankings,o){
let c=classification(b,rankings),
main=dominant(o),
attacker=bestAttackBoat(main);
if(c.type!=="cut"&&c.type!=="third")
return{revival:false,two:false,reason:""};
let dev=developmentBonus(b,rankings,main),
hs=hstats(b),
f=recentForm(b),
evidence=dev.reasons.length;
if(hs.count>=5&&hs.rate>=40)evidence++;
if(f.score>=3)evidence++;
if(strongMotor(b))evidence++;
if(dev.bonus>0&&evidence>=2)
return{
revival:true,
two:false,
reason:
`${attacker}号艇の${typeName(main)}展開。`+
`${dev.reasons.join("・")}。`+
`能力・過去15走・機力の根拠から3着まで残す`
};
return{revival:false,two:false,reason:""}
}
function thirdCandidates(a,secondBoat,rankings,type,firstBoat){
let o=overview(),
candidates=[],
mainAttacker=bestAttackBoat(type),
avg=scoreAverage(rankings);
for(let b=1;b<=6;b++){
if(b===firstBoat||b===secondBoat)continue;
let c=classification(b,rankings),
f=recentForm(b),
hs=hstats(b),
raw0=rankings.find(x=>x.boat===b)?.score||0,
raw=raw0,
reasons=[],
evidence=0;
if(raw>=avg-8){
evidence++;
reasons.push("AI評価")
}
if(suitability(b,type)>=15){
evidence++;
reasons.push("展開適性")
}
let dev=developmentBonus(b,rankings,type);
if(dev.bonus){
raw+=dev.bonus;
evidence++;
reasons.push(...dev.reasons)
}
if(promotionByMaki(b,rankings,type)){
raw+=12;
evidence++;
reasons.push("まくり後ろ2着昇格")
}
if(b===1&&o.escape>=50){
evidence++;
reasons.push("イン残り");
raw+=12
}
if(hs.count>=5&&hs.rate>=40){
evidence++;
reasons.push("同コース過去15走");
raw+=hscore(b)*.8
}
if(f.score>=3){
evidence++;
reasons.push("直近15走好調");
raw+=f.score*1.2
}
if(strongMotor(b)){
evidence++;
reasons.push("モーター");
raw+=8
}
if(evidence<2)continue;
if(c.type==="cut"){
let rv=attackRevival(b,rankings,o);
if(!rv.revival)continue;
candidates.push({
boat:b,
raw:raw+20,
revival:true,
thirdOnly:false,
two:false,
reason:rv.reason
});
continue
}
if(c.type==="third"){
candidates.push({
boat:b,
raw:raw+5,
revival:false,
thirdOnly:true,
two:promotionByMaki(b,rankings,type),
reason:
`${reasons.join("・")}の根拠あり`+
`${promotionByMaki(b,rankings,type)?" / まくり後ろ展開で2着まで昇格":""}`
});
continue
}
candidates.push({
boat:b,
raw,
revival:false,
thirdOnly:false,
two:promotionByMaki(b,rankings,type),
reason:`${reasons.join("・")}の根拠あり`
})
}
return candidates
.sort((a,b)=>b.raw-a.raw)
.slice(0,3)
}
function candidate(a,b,rankings,type,mode="recovery"){
if(a===b)return null;
let A=rankings.find(x=>x.boat===a),
B=rankings.find(x=>x.boat===b);
if(!A||!B)return null;
let o=overview(),
raw=A.score*1.35+B.score*.75;
if(type==="escape"){
if(course(a)!==1)return null;
raw+=30+o.escape*.9+suitability(a,type)*.7
}
if(type==="sashi"){
if(course(a)===1||course(b)!==2)return null;
raw+=10+suitability(a,type)*.9+o.sashi*.75+18;
if(o.escape<42)raw+=10
}
if(type==="maki"){
if(course(a)===1)return null;
raw+=8+suitability(a,type)*.9+o.maki*.75;
if(o.escape<42)raw+=10
}
if(type==="makiSashi"){
if(course(a)===1||course(a)===2)return null;
raw+=10+suitability(a,type)*.9+o.makiSashi*.75;
if(o.escape<42)raw+=10
}
if(type==="maki"&&course(a)===2){
let rr=decisions(racer(a));
if(rr.maki<10||o.maki<20)return null;
raw+=rr.maki*.5
}
if(b===1){
raw+=18;
if(o.escape>=50)raw+=10
}
raw+=
hscore(a)*1.3+
recentForm(a).score*1.5+
num(
racer(a)?.motor_top_2_percent||
racer(a)?.motor_2ren_rate||
racer(a)?.motor_2ren
)*.08;
let ra=rankings.findIndex(x=>x.boat===a)+1,
rb=rankings.findIndex(x=>x.boat===b)+1;
raw+=ra===1?8:ra===2?5:ra===3?3:0;
if(rb<=3)raw+=2;
let dev=developmentBonus(b,rankings,type);
raw+=dev.bonus;
if(
(type==="maki"||type==="makiSashi")&&
promotionByMaki(b,rankings,type)
)
raw+=10;
let vt=venueTrait();
raw*=
type==="escape"?vt.escape:
type==="sashi"?vt.sashi:
type==="maki"?vt.maki:
vt.makiSashi;
raw+=windSuitability(a,type)*1.2;
raw+=windSuitability(b,type)*.35;
if(windMode()==="head"&&windInfo().speed>=3){
if(type==="maki"&&course(a)>=3)raw+=10;
if(type==="makiSashi"&&course(a)>=3)raw+=7;
if(course(a)===1)raw-=5
}
if(windMode()==="tail"&&windInfo().speed>=3){
if(type==="escape"&&course(a)===1)raw+=10;
if(type==="sashi"&&course(a)===2)raw+=8
}
return{
key:`${a}-${b}`,
a,b,
raw:Math.max(.01,raw),
type
}
}
function pool(r,type,mode="recovery"){
let a=[];
for(let x=1;x<=6;x++)
for(let y=1;y<=6;y++){
let z=candidate(x,y,r,type,mode);
if(z)a.push(z)
}
return a.sort((a,b)=>b.raw-a.raw)
}

function startStrength(b){
let r=racer(b),p=preview(b),v=0;
let ast=num(r?.average_start_timing||r?.average_st),st=num(p.start_timing||p.st);
if(ast>0)v+=Math.max(0,.30-ast)*35;
if(st>0)v+=Math.max(0,.25-st)*30;
if(ast>0&&ast<=.15)v+=7;else if(ast>0&&ast<=.18)v+=4;
if(st>0&&st<=.15)v+=6;else if(st>0&&st<=.18)v+=3;
return v
}

}

}
/* 決まり手基本の動き：進入コース基準
   1=逃げ / 2=差し / 3=まくり / 4=差し / 5=まくり差し / 6=差し
   ただしST・機力・展示・本人実績で基本型を上回る決まり手は許容する */
function basicDecisionType(b){

function exhibitDecisionBonus(b,type){
initExhibitRatings();
let x=exhibitRatings[b]||{iki:0,nobi:0,turn:0,lap:0};
if(type==="maki")return x.iki*7+x.nobi*10+x.turn*3+x.lap*3;
if(type==="makiSashi")return x.iki*6+x.nobi*4+x.turn*9+x.lap*6;
if(type==="sashi")return x.iki*5+x.nobi*2+x.turn*9+x.lap*7;
return 0
}
function nonEscapeDecisionRanking(){
let d1=decisions(racer(1)),main=dominant(overview()),attacker=bestAttackBoat(main),list=[];
let weaknesses={
sashi:num(d1.sasareRate),
maki:num(d1.makareRate),
makiSashi:num(d1.makiSashiSareRate)
};
let maxWeak=Math.max(weaknesses.sashi,weaknesses.maki,weaknesses.makiSashi);
let weakTypes=Object.keys(weaknesses).filter(t=>weaknesses[t]===maxWeak);
for(let b=2;b<=6;b++){
let d=decisions(racer(b)),f=recentForm(b),motor=motorRate(b),basic=basicDecisionType(b);
let ownRates={sashi:num(d.sashi),maki:num(d.maki),makiSashi:num(d.makiSashi)};
let bestOwn=Math.max(ownRates.sashi,ownRates.maki,ownRates.makiSashi);
for(const type of ["sashi","maki","makiSashi"]){
let rate=ownRates[type],def=weaknesses[type],v=0,reasons=[];

/* ① 今回の1コース艇が何に弱いか：最重要 */
v+=def*1.85;
if(weakTypes.includes(type)&&maxWeak>0){v+=20;reasons.push("1コース最大弱点一致")}

/* ② 本人の得意な決まり手 */
v+=rate*1.65;
if(rate===bestOwn&&bestOwn>0){v+=16;reasons.push("本人得意")}

/* ③ コースごとの基本の動き */
if(type===basic){v+=18;reasons.push("基本の動き一致")}

/* ④ ST。速い艇ほど攻めを強化。まくり系はやや強め */
let st=startStrength(b);
v+=st*(type==="maki"?0.82:type==="makiSashi"?0.75:0.62);

/* ⑤ 機力・展示。展示は決まり手別に効かせる */
v+=motor*.30;
v+=exhibitDecisionBonus(b,type);
v+=f.score*2.2;

/* 総合能力は補助に留め、旧ロジックとの二重加点を避ける */
v+=score(b)*.10;

/* 今回の主攻め候補は小さく補助 */
if(b===attacker){v+=main===type?10:4;reasons.push("主攻め候補")}

/* 率0でも基本型・弱点一致・展示/STで展開上成立する場合は残す */
if(rate<=0&&type!==basic&&!weakTypes.includes(type)&&exhibitDecisionBonus(b,type)<=0)continue;

list.push({
boat:b,type,label:decisionTypeLabel(type),rate,def,motor,form:f.label,
attack:b===attacker,basic:type===basic,basicLabel:decisionTypeLabel(basic),
weak:weakTypes.includes(type),reason:reasons.join("・"),score:v
})
}
}
let sorted=list.sort((a,b)=>b.score-a.score),counts={},limited=[];
for(const x of sorted){
counts[x.boat]=(counts[x.boat]||0)+1;
if(counts[x.boat]<=2)limited.push(x);
if(limited.length>=10)break
}
return limited
}
