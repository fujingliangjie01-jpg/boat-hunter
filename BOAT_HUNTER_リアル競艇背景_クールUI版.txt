function makurareRateFromCourse(defenderBoat,attackerCourse){
  let a=history(defenderBoat);

    x.place!==1 &&
    x.againstDecision==="まくり" &&
    num(x.winnerCourse)===num(attackerCourse)
  ).length;
  return hit/a.length*100
}

  }

function ticketWeight(t,rankings){
  let pos=new Map(rankings.map((x,i)=>[x.boat,{i,score:Math.max(1,x.score)}]));
  let w=1;
  t.legs.forEach((b,j)=>{
    let x=pos.get(b),base=x?Math.max(1,7-x.i):1;
    w*=base*(j===0?1.35:j===1?1.1:1)
  });
  if(t.head!==1){
    let e=safeNonEscapeDecisionRanking().find(x=>x.boat===t.head&&x.label===t.decision);
    if(e)w*=1+Math.max(0,e.score)/250
  }
  return w
}
function addCombos(out,pattern,firsts,seconds,thirds,decisionEntry,label,isTwo=false){
  firsts=[...new Set(firsts.filter(Boolean))];seconds=[...new Set(seconds.filter(Boolean))];
  thirds=[...new Set((thirds||[]).filter(Boolean))];
  for(let a of firsts)for(let b of seconds){
    if(a===b)continue;
    if(isTwo){out.push({legs:[a,b],pattern,label,decision:a===1?"逃げ":(decisionEntry?.label||"-"),head:a});continue}
    for(let c of thirds){
      if(c===a||c===b)continue;
      out.push({legs:[a,b,c],pattern,label,decision:a===1?"逃げ":(decisionEntry?.label||"-"),head:a})
    }
  }
}

function finalizeTickets(base,rankings){
  /* 各1着の決まり手必須買い目を先に保護。同一買い目は統合 */
  let mandatory=[],heads=new Map();
  base.forEach(t=>{
    if(t.head===1)return;
    /* 2連単の頭は決まり手必須3連単を発生させない */
    if(t.label==="2連単"||t.legs?.length===2)return;
    let e=safeNonEscapeDecisionRanking().find(x=>x.boat===t.head&&x.label===t.decision);
    if(e&&!heads.has(e.boat+":"+e.type))heads.set(e.boat+":"+e.type,e)
  });
  heads.forEach(e=>mandatory.push(...mandatoryTicketsFor(e)));
  let all=[...mandatory,...base],map=new Map();
  all.forEach(t=>{
    if(new Set(t.legs).size!==t.legs.length)return;
    let k=t.legs.join("-");
    if(!map.has(k))map.set(k,{...t});
    else if(t.mandatory)map.set(k,{...map.get(k),mandatory:true,decision:t.decision,head:t.head})
  });
  let arr=[...map.values()];
  arr.forEach(t=>t.weight=ticketWeight(t,rankings));
  let total=arr.reduce((z,t)=>z+t.weight,0)||1;
  arr.forEach(t=>t.pct=t.weight/total*100);
  let must=arr.filter(t=>t.mandatory).sort((a,b)=>b.pct-a.pct);
  let normal=arr.filter(t=>!t.mandatory).sort((a,b)=>b.pct-a.pct);
  /* 原則20点。必須が20点を超えた場合は必須を全て残す */
  let keep=[...must];
  if(keep.length<20)keep.push(...normal.slice(0,20-keep.length));
  keep.sort((a,b)=>b.pct-a.pct);
  return keep
}

function startStrength(b){
let r=racer(b),p=preview(b),v=0;
let ast=num(r?.average_start_timing||r?.average_st),st=num(p.start_timing||p.st);
if(ast>0)v+=Math.max(0,.30-ast)*35;
if(st>0)v+=Math.max(0,.25-st)*30;
if(ast>0&&ast<=.15)v+=7;else if(ast>0&&ast<=.18)v+=4;
if(st>0&&st<=.15)v+=6;else if(st>0&&st<=.18)v+=3;
return v
}

function escapeDecisionRanking(rankings){

if(b===attacker)v+=12;

}
/* 決まり手基本の動き：進入コース基準
   1=逃げ / 2=差し / 3=まくり / 4=差し / 5=まくり差し / 6=差し
   ただしST・機力・展示・本人実績で基本型を上回る決まり手は許容する */
function basicDecisionType(b){
let c=course(b);
return{1:"escape",2:"sashi",3:"maki",4:"sashi",5:"makiSashi",6:"sashi"}[c]||"sashi"
}

}
function exhibitDecisionBonus(b,type){
initExhibitRatings();
let x=exhibitRatings[b]||{iki:0,nobi:0,turn:0,lap:0};
if(type==="maki")return x.iki*7+x.nobi*10+x.turn*3+x.lap*3;
if(type==="makiSashi")return x.iki*6+x.nobi*4+x.turn*9+x.lap*6;
if(type==="sashi")return x.iki*5+x.nobi*2+x.turn*9+x.lap*7;
return 0
}
function nonEscapeDecisionRanking(){
let d1=decisions(racer(1)),main=dominant(overview()),attacker=bestAttackBoat(main),list=[];
let aiTop4=new Set(ranking().slice(0,4).map(x=>x.boat));

[["sashi","差し","sasareRate"],["maki","まくり","makareRate"],["makiSashi","まくり差し","makiSashiSareRate"]]
.forEach(([type,label,defKey])=>{
  for(let b=2;b<=6;b++){
    /* 加点合計を出しても、AI期待値ランキング4位以内の艇だけ表示対象 */
    if(!aiTop4.has(b))continue;

    let d=decisions(racer(b));
    let rate=type==="sashi"?num(d.sashi):type==="maki"?num(d.maki):num(d.makiSashi);
    let motor=motorRate(b),def=0,wallParts=[];

    if(type==="maki"){
      let link=makiDefenseLink(b);
      def=link.total;
      wallParts=link.parts;
    }else{
      def=num(d1[defKey]);
    }

    /* 新・決まり手評価
       直近調子スコアは加点しない */
    let v=
      rate*1.7+
      motor*.28+
      startStrength(b)*.6+
      score(b)*.16+
      suitability(b,type)*.42;

    /* 被攻め連動 */
    if(type==="maki"){
      /* 2号艇=1号艇、3号艇=1+2、4〜6号艇=1+2+3 を大きめ加点 */
      v+=def*.8;
    }else{
      /* 差し・まくり差しは1号艇の対応弱点を維持 */
      v+=def*1.05;
    }

    if(b===attacker)v+=main===type?16:8;

    list.push({
      boat:b,type,label,rate,def,motor,
      attack:b===attacker,
      wallParts,
      score:v
    })
  }
});

let pool=list.sort((a,b)=>b.score-a.score);
let limited=[],counts={},lastPos={};

/* 同一艇は最大2件。
   同一艇2件目は、1件目との間に最低2件の別艇を挟む。
   例：1位と同じ艇の2件目は最短4位。 */
while(pool.length&&limited.length<10){
  let pickIndex=-1;
  for(let i=0;i<pool.length;i++){
    let x=pool[i];
    let count=counts[x.boat]||0;
    if(count>=2)continue;
    let prev=lastPos[x.boat];
    if(prev!=null && limited.length-prev<3)continue;
    pickIndex=i;
    break
  }
  if(pickIndex<0)break;
  let x=pool.splice(pickIndex,1)[0];
  counts[x.boat]=(counts[x.boat]||0)+1;
  lastPos[x.boat]=limited.length;
  limited.push(x)
}
return limited
}
